/// <summary>
/// Pushes changes made in SQL Server up to Business Central: EQData.dbo.Customer
/// drives both the BC customer and the BC project that share its number.
///
/// Customers are written through the standard v2.0 API. Projects are written through
/// AlphaBOLD's custom projects API (table 167), because the standard projects entity
/// exposes only number and displayName.
///
/// This is the first job in the codebase that WRITES to BC, so the safety model is
/// different from the read-only syncs. A bad read is corrected on the next run; a bad
/// write lands in the system of record and stays there.
///
/// Four consequences, all deliberate:
///
///   1. SCOPED BY CREATE DATE. EQData.dbo.Customer holds 15,453 rows and BC has 2,862
///      customers, so without a filter roughly 12,500 rows would have no BC counterpart
///      and every one would look like a create. BcSync:CustomerPushCreatedAfter bounds
///      it -- 2026-01-01 gives ~348 rows, close to what BC already has. It is config so
///      the scope can be widened without a deploy.
///
///   2. CREATES ARE GATED SEPARATELY. BcSync:CustomerPushAllowCreate defaults to false,
///      so updates can run live while creates stay off. A bad update is corrected next
///      run; a bad create leaves a duplicate nobody notices until an invoice goes to the
///      wrong record.
///
///   3. DRY RUN BY DEFAULT. BcSync:CustomerPushDryRun defaults to true: the job does
///      the full comparison and logs the exact payload it would send, but sends nothing.
///      Set it to false only once the logged output looks right.
///
///   4. COMPARE BEFORE WRITING. The watermark picks the candidates; the value
///      comparison decides what actually goes. Without that second step, a PATCH would
///      bump BC's lastModifiedDateTime, the pull would see "changed", write it back to
///      SQL, which bumps ModifiedDateTime, which re-queues the push -- a loop that
///      never settles even though nothing ever changes. Comparing values breaks it:
///      a record already in agreement produces no write however many timestamps moved.
/// </summary>

    /// <summary>
    /// SalesmanOne holds surnames that mostly match BC's salesperson codes directly --
    /// 14 of the 16 distinct values do. These three do not: two are typos of EXECUTIVE,
    /// and POBANZ exists in BC only under the variant POBANZ-J.
    ///
    /// salespersonCode is a REFERENCE: BC validates it against its Salesperson/Purchaser
    /// list and rejects the whole write if it does not resolve. Anything not fixed up
    /// here is still sent as-is and will fail loudly for that one record rather than
    /// being silently dropped, which is the behaviour that surfaces a new salesperson.
    /// </summary>
    private static readonly Dictionary<string, string> SalespersonFixups =
        new(StringComparer.OrdinalIgnoreCase)
        {
            ["EXCECUTIVE"] = "EXECUTIVE",
            ["EXEUTIVE"] = "EXECUTIVE",
            ["POBANZ"] = "POBANZ-J",
        };

    private static string? MapSalesperson(string? raw)
    {
        var s = raw?.Trim();
        if (string.IsNullOrEmpty(s)) return null;
        return SalespersonFixups.TryGetValue(s, out var fixedUp) ? fixedUp : s;
    }

    /// <summary>
    /// ProjectEmail1 sometimes holds several addresses separated by commas, and BC's
    /// email field takes one. Takes the first and leaves the rest -- which is lossy, but
    /// the alternative is BC rejecting the value outright.
    ///
    /// Worth knowing this is the PROJECT contact, not the customer's own address: two
    /// different companies on the same job can share it.
    /// </summary>
    private static string? FirstEmail(string? raw)
    {
        var s = raw?.Trim();
        if (string.IsNullOrEmpty(s)) return null;
        var first = s.Split(',', StringSplitOptions.RemoveEmptyEntries).FirstOrDefault()?.Trim();
        return string.IsNullOrEmpty(first) ? null : first;
    }

    private static string BcDate(DateTime d) => d.ToString("yyyy-MM-dd", CultureInfo.InvariantCulture);

    private static bool IsBcEmpty(string? value) =>
        string.IsNullOrWhiteSpace(value) || value == BcEmptyDate;

    /// <summary>
    /// Class and model, e.g. "HAMMERHEAD 550 EC-H 20 LITRONIC". Uppercased because BC
    /// stores Search Description uppercased; a mixed-case value would compare as
    /// different on every run.
    /// </summary>
    private static string? SearchDescription(string? cls, string? model)
    {
        var s = string.Join(" ", new[] { cls?.Trim(), model?.Trim() }.Where(p => !string.IsNullOrEmpty(p)));
        return s.Length == 0 ? null : s.ToUpperInvariant();
    }

    /// <summary>
    /// "550 EC-H 20 Litronic S/N 55.278". Just the model when there is no serial, and
    /// nothing when there is no model.
    /// </summary>
    private static string? YourReference(string? model, string? serial)
    {
        var m = model?.Trim();
        if (string.IsNullOrEmpty(m)) return null;
        var s = serial?.Trim();
        return string.IsNullOrEmpty(s) ? m : $"{m} S/N {s}";
    }

    /// <summary>
    /// "NET 30" becomes "NET30". Only returns a code BC actually has: paymentTermsCode
    /// is a reference, and BC rejects the whole PATCH for a code it does not know. An
    /// unknown value is logged once per run and BC's value is left alone.
    /// </summary>
    private string? MapPaymentTerms(string? raw, IReadOnlySet<string> codes, HashSet<string> warned)
    {
        var s = raw?.Trim();
        if (string.IsNullOrEmpty(s)) return null;

        var code = s.Replace(" ", "").ToUpperInvariant();
        if (codes.Contains(code)) return code;

        if (warned.Add(s))
            _logger.LogWarning(
                "Customer push: payment terms '{Terms}' has no BC payment terms code '{Code}'; left unchanged in BC.",
                s, code);
        return null;
    }

    // 30 minutes: two runs with overlapping candidate sets would both PATCH the same
    // records, and the second would fail on If-Match having read a now-stale etag.
    [DisableConcurrentExecution(30 * 60)]
    public async Task RunAsync(CancellationToken ct)
    {
        var target = BcTarget.Resolve(_bc, _config, JobKey);
        var dryRun = _config.GetValue<bool?>("BcSync:CustomerPushDryRun") ?? true;
        var allowCreate = _config.GetValue<bool?>("BcSync:CustomerPushAllowCreate") ?? false;

        // Scope bound. Config rather than hardcoded so it can be widened without a
        // deploy: 2026-01-01 gives ~348 rows, 2025 gives 754, 2024 gives 1,046.
        var createdAfter = _config.GetValue<DateTime?>("BcSync:CustomerPushCreatedAfter")
            ?? new DateTime(2026, 1, 1);

        _logger.LogInformation(
            "Customer push: targeting environment {Environment}, company {CompanyId}. " +
            "DryRun={DryRun}, AllowCreate={AllowCreate}, CreatedAfter={CreatedAfter:yyyy-MM-dd}.",
            target.Environment, target.CompanyId, dryRun, allowCreate, createdAfter);

        var runId = await _runLog.StartAsync(JobKey, target, ct);
        try
        {
            EnsureCustomApiMatches(target);

            var since = await _watermarks.GetAsync(JobKey, ct);

            // --- Candidates: rows a user (or anything else) touched since last run,
            // bounded by create date. The trigger on EQData.dbo.Customer stamps
            // ModifiedDateTime on every write path, so this catches app edits, stored
            // procedures and ad-hoc SQL alike.
            var candidates = (await ReadCandidatesAsync(since, createdAfter, ct)).ToList();

            if (candidates.Count == 0)
            {
                _logger.LogInformation("Customer push: nothing changed since {Since:o}.", since);
                await _runLog.FinishAsync(runId, new SyncRunCounts
                {
                    WatermarkFrom = since,
                    Fetched = 0,
                    HeadersChanged = 0,
                    LinesChanged = 0
                }, null, ct);
                return;
            }

            // Captured BEFORE any write-back. Writing BC's id back into the source table
            // bumps ModifiedDateTime via the trigger, so anchoring the new watermark to
            // what we READ means those rows come round once more, find no difference and
            // settle. One wasted comparison per written row; the alternative (anchoring
            // after the write-back) would silently skip a user edit made during the run.
            var newWatermark = new DateTimeOffset(
                candidates.Max(c => c.ModifiedDateTime), TimeSpan.Zero);

            // --- Fetch BC's current state for just these numbers, in batches.
            var numbers = candidates.Select(c => c.CustomerID).ToList();
            var bcCustomers = await FetchByNumberAsync(target, "customers", numbers, ct);
            var paymentTermCodes = await FetchPaymentTermCodesAsync(target, ct);
            var projectFields = BuildProjectFields(
                paymentTermCodes, new HashSet<string>(StringComparer.OrdinalIgnoreCase));
            var bcProjects = await FetchProjectsAsync(numbers, projectFields, ct);

            _logger.LogInformation(
                "Customer push: {Candidates} changed rows; BC returned {Customers} customers, {Projects} projects.",
                candidates.Count, bcCustomers.Count, bcProjects.Count);

            var customersCreated = 0;
            var projectsCreated = 0;
            var createFailed = 0;
            var customersPatched = 0;
            var projectsPatched = 0;
            var skippedMissing = 0;
            var conflicts = 0;
            var projectsFailed = 0;

            foreach (var row in candidates)
            {
                // --- Customer
                var customerExists = bcCustomers.TryGetValue(row.CustomerID, out var cust);

                if (!customerExists)
                {
                    if (!allowCreate)
                    {
                        _logger.LogWarning(
                            "Customer push: {Number} has no BC customer; creates are disabled.", row.CustomerID);
                        skippedMissing++;
                    }
                    else
                    {
                        // number is supplied rather than left to a BC number series. That is
                        // load-bearing: it means a retry after a lost response hits BC's
                        // uniqueness check and fails, instead of silently creating a second
                        // customer. Never let BC assign the number here.
                        var body = BuildCreate(CustomerFields, row);
                        body["number"] = row.CustomerID;

                        var newId = await PostAsync(target, "customers", body, dryRun, row.CustomerID, ct);
                        if (newId is not null)
                        {
                            customersCreated++;
                            customerExists = true;
                            await StoreBcIdAsync(row.CustomerID, newId.Value, ct);
                        }
                        else if (dryRun)
                        {
                            // Nothing was created, so there is no id -- but the point of a dry
                            // run is to see every payload, so carry on to the project below as
                            // though the customer were present.
                            customersCreated++;
                            customerExists = true;
                        }
                        else
                        {
                            // Live run: the project would fail anyway if it needs the customer.
                            createFailed++;
                            continue;
                        }
                    }
                }
                else
                {
                    var patch = BuildPatch(CustomerFields, row, f => f.BcField switch
                    {
                        "displayName" => cust.DisplayName,
                        "addressLine1" => cust.AddressLine1,
                        "addressLine2" => cust.AddressLine2,
                        "city" => cust.City,
                        "state" => cust.State,
                        "postalCode" => cust.PostalCode,
                        "country" => cust.Country,
                        "phoneNumber" => cust.PhoneNumber,
                        "email" => cust.Email,
                        "salespersonCode" => cust.SalespersonCode,
                        _ => throw new InvalidOperationException($"No BC reader for customer field '{f.BcField}'.")
                    });

                    if (patch.Count > 0)
                    {
                        var ok = await PatchAsync(target, "customers", cust.Id, cust.Etag, patch, dryRun, row.CustomerID, ct);
                        if (ok is PatchResult.Written or PatchResult.DryRun) customersPatched++;
                        else if (ok == PatchResult.Conflict) conflicts++;
                    }
                }

                // --- Project, through the custom API. Deliberately after the customer: on
                // a create run the customer above may have only just been made, and the
                // project is linked to it.
                if (!bcProjects.TryGetValue(row.CustomerID, out var proj))
                {
                    if (!allowCreate)
                    {
                        _logger.LogWarning(
                            "Customer push: {Number} has no BC project; creates are disabled.", row.CustomerID);
                    }
                    else if (customerExists)
                    {
                        var body = BuildCreate(projectFields, row);
                        body["number"] = row.CustomerID;

                        // Links the project to its customer. BC copies the customer's
                        // details onto the project when this is set; if it applies that
                        // after the fields above, any difference is corrected the next
                        // time this row is compared.
                        body["sellToCustomerNo"] = row.CustomerID;

                        if (await CreateProjectAsync(body, dryRun, row.CustomerID, ct)) projectsCreated++;
                        else createFailed++;
                    }
                }
                else
                {
                    var patch = BuildPatch(projectFields, row, f => proj.Values.GetValueOrDefault(f.BcField));

                    if (patch.Count > 0)
                    {
                        var ok = await PatchProjectAsync(proj, patch, dryRun, row.CustomerID, ct);
                        if (ok is PatchResult.Written or PatchResult.DryRun) projectsPatched++;
                        else if (ok == PatchResult.Conflict) conflicts++;
                        else projectsFailed++;
                    }
                }
            }

            // Only advance the watermark on a real run. A dry run that moved it would
            // mean the first live run had nothing left to send.
            if (!dryRun)
                await _watermarks.SetAsync(JobKey, newWatermark, ct);

            await _runLog.FinishAsync(runId, new SyncRunCounts
            {
                WatermarkFrom = since,
                WatermarkTo = dryRun ? null : newWatermark,
                Fetched = candidates.Count,
                HeadersChanged = customersPatched + customersCreated,
                LinesChanged = projectsPatched + projectsCreated,
                // Per-record write failures of either kind: a create BC refused, or a
                // project PATCH BC rejected as invalid.
                CreateFailed = createFailed + projectsFailed,
                Conflicts = conflicts,
                SkippedMissing = skippedMissing
            }, null, ct);

            _logger.LogInformation(
                "Customer push{DryRun}: {Candidates} candidates. Customers {CustUpd} updated, {CustNew} created. " +
                "Projects {ProjUpd} updated, {ProjNew} created, {ProjFailed} rejected. {Missing} skipped, {Failed} create failures, {Conflicts} conflicts.",
                dryRun ? " (DRY RUN)" : "", candidates.Count,
                customersPatched, customersCreated, projectsPatched, projectsCreated, projectsFailed,
                skippedMissing, createFailed, conflicts);
        }
        catch (Exception ex)
        {
            await _runLog.FinishAsync(runId, new SyncRunCounts(), ex.ToString(), CancellationToken.None);
            throw;
        }
    }

    // ---------------------------------------------------------------- SQL

    // Column order must match the SELECT below: Dapper maps rows onto this constructor
    // by position as well as name.
    private sealed record Candidate(
        string CustomerID, string? Name, string? ProjectName,
        string? MailingAddress, string? MailingTelephone, string? TaxingCountry,
        string? ProjectEmail1, string? SalesmanOne,
        string? MailingContact, string? Class, string? Model, string? SerialNumber,
        string? ShippingContact, string? ShippingTelephone,
        DateTime? ErectionDate, DateTime? DismantleDate, string? PMTTerms,
        DateTime ModifiedDateTime);

    private async Task<IEnumerable<Candidate>> ReadCandidatesAsync(
        DateTimeOffset since, DateTime createdAfter, CancellationToken ct)
    {
        // CreateDate bounds the scope: without it, the ~12,500 source rows with no BC
        // counterpart would every one look like a create. ModifiedDateTime is maintained
        // by trgCustomer_ModifiedDateTime on the source table, so this catches edits made
        // through the Alpha dialog, the helper functions, the stored procedures and
        // ad-hoc SQL alike -- RevisionDate does not, which is why that column is not used.
        //
        // Email, salesperson and payment terms live in two other tables, joined the same way
        // v_eqcCustomerEdit does. Both are LEFT joins: a customer with no billing
        // instructions or supplementary record still syncs, just without those fields.
        //
        // NOTE the watermark only tracks EQData.dbo.Customer. An edit to CustSupp or
        // BillingInstructions alone will not re-queue the customer, because neither
        // carries the trigger. Those changes reach BC when the customer row is next
        // touched for any reason.
        var sql = $@"
SELECT c.[CustomerID], c.[Name], c.[ProjectName], c.[MailingAddress], c.[MailingTelephone],
       c.[TaxingCountry], s.[ProjectEmail1], b.[SalesmanOne],
       c.[MailingContact], c.[Class], c.[Model], c.[SerialNumber],
       c.[ShippingContact], c.[ShippingTelephone],
       c.[ErectionDate], c.[DismantleDate], b.[PMTTerms],
       c.[ModifiedDateTime]
  FROM {SourceTable} c
  LEFT JOIN EQData.dbo.CustSupp s ON s.[CustomerID] = c.[CustomerID]
  LEFT JOIN EQDataCor.dbo.BillingInstructions b ON b.[CustomerID] = c.[CustomerID]
 WHERE ISNULL(c.[Inactive], 0) = 0
   -- A row with no number or no name cannot be pushed: BC rejects it with a
   -- displayName-must-be-provided error. Three such rows exist and would otherwise
   -- fail on every single run, which trains people to ignore the failure count.
   AND NULLIF(LTRIM(RTRIM(c.[CustomerID])), '') IS NOT NULL
   AND NULLIF(LTRIM(RTRIM(c.[Name])), '') IS NOT NULL
   AND c.[CreateDate] >= @CreatedAfter
   AND c.[ModifiedDateTime] > @Since
 ORDER BY c.[ModifiedDateTime];";

        using var conn = new SqlConnection(_connectionString);
        return await conn.QueryAsync<Candidate>(
            new CommandDefinition(sql, new { Since = since, CreatedAfter = createdAfter }, cancellationToken: ct));
    }

    // ---------------------------------------------------------------- BC reads

    private sealed record BcRecord(Guid Id, string Number, string? DisplayName,
        string? AddressLine1, string? AddressLine2, string? City, string? State,
        string? PostalCode, string? Country, string? PhoneNumber, string? Email,
        string? SalespersonCode, string Etag);

    /// <summary>
    /// Fetches only the records we might write, keyed by number. Batched because a
    /// filter is a URL and a few hundred "number eq 'x' or ..." clauses would exceed
    /// what BC accepts.
    /// </summary>
    private async Task<Dictionary<string, BcRecord>> FetchByNumberAsync(
        BcTarget target, string entity, IReadOnlyList<string> numbers, CancellationToken ct)
    {
        const int batchSize = 20;
        var result = new Dictionary<string, BcRecord>(StringComparer.OrdinalIgnoreCase);
        var http = _httpFactory.CreateClient();

        for (var i = 0; i < numbers.Count; i += batchSize)
        {
            var batch = numbers.Skip(i).Take(batchSize).ToList();
            var filter = string.Join(" or ", batch.Select(n => $"number eq '{n.Replace("'", "''")}'"));
            var url = $"{target.ApiBase}/{entity}?$filter={Uri.EscapeDataString(filter)}";

            var token = await _tokens.GetTokenAsync(ct);
            using var req = new HttpRequestMessage(HttpMethod.Get, url);
            req.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);

            using var res = await http.SendAsync(req, ct);
            var body = await res.Content.ReadAsStringAsync(ct);
            if (!res.IsSuccessStatusCode)
                throw new HttpRequestException($"BC {entity} fetch failed ({(int)res.StatusCode}): {body}");

            using var doc = JsonDocument.Parse(body);
            if (!doc.RootElement.TryGetProperty("value", out var arr)) continue;

            foreach (var el in arr.EnumerateArray())
            {
                var number = Str(el, "number");
                var id = Str(el, "id");
                if (number is null || id is null || !Guid.TryParse(id, out var guid)) continue;

                result[number] = new BcRecord(
                    guid, number,
                    Str(el, "displayName"),
                    Str(el, "addressLine1"),
                    Str(el, "addressLine2"),
                    Str(el, "city"),
                    Str(el, "state"),
                    Str(el, "postalCode"),
                    Str(el, "country"),
                    Str(el, "phoneNumber"),
                    Str(el, "email"),
                    Str(el, "salespersonCode"),
                    Str(el, "@odata.etag") ?? "*");
            }
        }

        return result;
    }

    // ---------------------------------------------------------------- custom API (projects)

    /// <summary>
    /// Projects are written through the custom API and customers through the standard
    /// one, which are configured separately. If they pointed at different companies, a
    /// run would update a customer in one place and its project in another, so the run
    /// refuses to start instead.
    /// </summary>
    private void EnsureCustomApiMatches(BcTarget target)
    {
        if (!_customApi.IsConfigured)
            throw new InvalidOperationException(
                "Customer push writes projects through the custom API, but the BcCustomApi section " +
                "is not fully configured. Check TenantId, Environment, CompanyId and PathSegment.");

        var s = _customApi.Settings;
        var sameCompany =
            string.Equals(s.TenantId, target.TenantId, StringComparison.OrdinalIgnoreCase) &&
            string.Equals(s.Environment, target.Environment, StringComparison.OrdinalIgnoreCase) &&
            Guid.TryParse(s.CompanyId, out var customCompany) &&
            Guid.TryParse(target.CompanyId, out var targetCompany) &&
            customCompany == targetCompany;

        if (!sameCompany)
            throw new InvalidOperationException(
                $"Customer push targets {target.Environment}/{target.CompanyId}, but BcCustomApi points at " +
                $"{s.Environment}/{s.CompanyId}. Customers and projects must be written to the same company.");
    }

    /// <summary>A project as BC holds it: its etag plus the current value of every
    /// field in the project field map, keyed by BC field name.</summary>
    private sealed record BcProject(string Number, string Etag, Dictionary<string, string?> Values);

    private async Task<Dictionary<string, BcProject>> FetchProjectsAsync(
        IReadOnlyList<string> numbers, PushField[] fields, CancellationToken ct)
    {
        const int batchSize = 20;
        var result = new Dictionary<string, BcProject>(StringComparer.OrdinalIgnoreCase);

        for (var i = 0; i < numbers.Count; i += batchSize)
        {
            var filter = string.Join(" or ", numbers.Skip(i).Take(batchSize)
                .Select(n => $"number eq '{n.Replace("'", "''")}'"));

            foreach (var el in await _customApi.FetchAllAsync("projects", filter, ct))
            {
                var number = Str(el, "number");
                if (number is null) continue;

                // Every mapped field comes back as a JSON string, dates and enums included.
                var values = fields.ToDictionary(f => f.BcField, f => Str(el, f.BcField));
                result[number] = new BcProject(number, Str(el, "@odata.etag") ?? "*", values);
            }
        }

        return result;
    }

    /// <summary>The payment terms codes BC has, so the push never sends one it would
    /// reject. Read through the standard API, which our own app can reach.</summary>
    private async Task<IReadOnlySet<string>> FetchPaymentTermCodesAsync(BcTarget target, CancellationToken ct)
    {
        var http = _httpFactory.CreateClient();
        var token = await _tokens.GetTokenAsync(ct);

        using var req = new HttpRequestMessage(HttpMethod.Get, $"{target.ApiBase}/paymentTerms");
        req.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);

        using var res = await http.SendAsync(req, ct);
        var body = await res.Content.ReadAsStringAsync(ct);
        if (!res.IsSuccessStatusCode)
            throw new HttpRequestException($"BC paymentTerms fetch failed ({(int)res.StatusCode}): {body}");

        var codes = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        using var doc = JsonDocument.Parse(body);
        if (doc.RootElement.TryGetProperty("value", out var arr))
            foreach (var el in arr.EnumerateArray())
                if (Str(el, "code") is { Length: > 0 } code)
                    codes.Add(code);

        return codes;
    }

    private static bool IsSuccess(HttpStatusCode status) => (int)status is >= 200 and < 300;

    private async Task<bool> CreateProjectAsync(
        Dictionary<string, object?> body, bool dryRun, string number, CancellationToken ct)
    {
        var json = JsonSerializer.Serialize(body);

        if (dryRun)
        {
            _logger.LogInformation("DRY RUN would POST projects {Number}: {Payload}", number, json);
            return true;
        }

        var (status, responseBody) = await _customApi.SendAsync(HttpMethod.Post, "projects", json, null, ct);
        if (!IsSuccess(status))
        {
            // Not fatal to the run, same as a customer create.
            _logger.LogError(
                "POST projects {Number} failed ({Status}): {Body}", number, (int)status, responseBody);
            return false;
        }

        _logger.LogInformation("Created projects {Number}: {Payload}", number, json);
        return true;
    }

    /// <summary>
    /// PATCHes a project. The custom API's key is the project number, so the URL is
    /// projects('17065') rather than the standard API's projects(guid).
    ///
    /// Dates are the one ordering hazard. BC rejects a starting date later than the
    /// ending date, checking each field against the value already stored. If both dates
    /// move forward past the current ending date, applying the starting date first
    /// would fail, so in that case the ending date goes in its own PATCH first.
    /// </summary>
    private async Task<PatchResult> PatchProjectAsync(
        BcProject proj, Dictionary<string, object?> patch, bool dryRun, string number, CancellationToken ct)
    {
        if (dryRun)
        {
            _logger.LogInformation(
                "DRY RUN would PATCH projects {Number}: {Payload}", number, JsonSerializer.Serialize(patch));
            return PatchResult.DryRun;
        }

        var path = $"projects('{Uri.EscapeDataString(number.Replace("'", "''"))}')";
        var etag = proj.Etag;

        if (EndingDateMustGoFirst(proj, patch))
        {
            var first = new Dictionary<string, object?> { ["endingDate"] = patch["endingDate"] };
            patch.Remove("endingDate");

            var (firstResult, newEtag) = await SendProjectPatchAsync(path, etag, first, number, ct);
            if (firstResult != PatchResult.Written) return firstResult;

            // We wrote this record a moment ago, so if BC returned no etag "*" is safe.
            etag = newEtag ?? "*";
        }

        var (result, _) = await SendProjectPatchAsync(path, etag, patch, number, ct);
        return result;
    }

    private static bool EndingDateMustGoFirst(BcProject proj, Dictionary<string, object?> patch)
    {
        if (!patch.TryGetValue("startingDate", out var newStart) || !patch.ContainsKey("endingDate"))
            return false;

        var currentEnd = proj.Values.GetValueOrDefault("endingDate");
        if (IsBcEmpty(currentEnd)) return false;   // BC skips the check when there is no ending date

        // yyyy-MM-dd strings compare correctly as text.
        return string.CompareOrdinal(newStart as string, currentEnd) > 0;
    }

    /// <summary>
    /// One PATCH. A 412 is a conflict, as for customers. A 400 is BC rejecting this
    /// record's data (a validation error): logged and counted, and the run carries on,
    /// so one bad record cannot block every other push. Anything else non-successful
    /// (auth, throttling, BC down) throws, so the watermark is not advanced and Hangfire
    /// retries the run.
    /// </summary>
    private async Task<(PatchResult Result, string? NewEtag)> SendProjectPatchAsync(
        string path, string etag, Dictionary<string, object?> patch, string number, CancellationToken ct)
    {
        var json = JsonSerializer.Serialize(patch);
        var (status, body) = await _customApi.SendAsync(HttpMethod.Patch, path, json, etag, ct);

        if (status == HttpStatusCode.PreconditionFailed)
        {
            _logger.LogWarning(
                "Customer push: projects {Number} changed in BC since we read it (412); skipped this run.", number);
            return (PatchResult.Conflict, null);
        }

        if (status == HttpStatusCode.BadRequest)
        {
            _logger.LogError(
                "PATCH projects {Number} rejected by BC (400). Sent {Payload}; BC said {Body}", number, json, body);
            return (PatchResult.Failed, null);
        }

        if (!IsSuccess(status))
            throw new HttpRequestException($"PATCH projects {number} failed ({(int)status}): {body}");

        _logger.LogInformation("Patched projects {Number}: {Payload}", number, json);

        string? newEtag = null;
        try
        {
            if (!string.IsNullOrWhiteSpace(body))
            {
                using var doc = JsonDocument.Parse(body);
                newEtag = Str(doc.RootElement, "@odata.etag");
            }
        }
        catch (JsonException) { /* no usable body; the caller falls back to "*" */ }

        return (PatchResult.Written, newEtag);
    }

    // ---------------------------------------------------------------- BC writes

    private enum PatchResult { Written, DryRun, Conflict, Failed }

    private async Task<PatchResult> PatchAsync(
        BcTarget target, string entity, Guid id, string etag,
        Dictionary<string, object?> patch, bool dryRun, string number, CancellationToken ct)
    {
        var json = JsonSerializer.Serialize(patch);

        if (dryRun)
        {
            _logger.LogInformation("DRY RUN would PATCH {Entity} {Number}: {Payload}", entity, number, json);
            return PatchResult.DryRun;
        }

        var http = _httpFactory.CreateClient();
        var token = await _tokens.GetTokenAsync(ct);

        using var req = new HttpRequestMessage(HttpMethod.Patch, $"{target.ApiBase}/{entity}({id})");
        req.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
        // BC requires If-Match on PATCH. A 412 back means someone edited this record in
        // BC between our read and our write -- the conflict signal. We log and move on
        // rather than overwriting: the next run re-reads and re-compares.
        req.Headers.TryAddWithoutValidation("If-Match", etag);
        req.Content = new StringContent(json, Encoding.UTF8, "application/json");

        using var res = await http.SendAsync(req, ct);
        var body = await res.Content.ReadAsStringAsync(ct);

        if (res.StatusCode == HttpStatusCode.PreconditionFailed)
        {
            _logger.LogWarning(
                "Customer push: {Entity} {Number} changed in BC since we read it (412); skipped this run.",
                entity, number);
            return PatchResult.Conflict;
        }

        if (!res.IsSuccessStatusCode)
            throw new HttpRequestException(
                $"PATCH {entity} {number} failed ({(int)res.StatusCode}): {body}");

        _logger.LogInformation("Patched {Entity} {Number}: {Payload}", entity, number, json);
        return PatchResult.Written;
    }

    // ---------------------------------------------------------------- helpers

    /// <summary>
    /// Builds the PATCH body from a field map: one entry per field whose SQL value
    /// genuinely differs from BC's. An empty result means no request at all, which is
    /// what stops a write echoing back through the pull.
    /// </summary>
    private Dictionary<string, object?> BuildPatch(
        PushField[] fields, Candidate row, Func<PushField, string?> readBc)
    {
        var patch = new Dictionary<string, object?>();

        foreach (var f in fields)
        {
            var ours = Clip(f.Read(row), f.MaxLength, row.CustomerID, f.BcField);
            var theirs = readBc(f);

            if (string.IsNullOrWhiteSpace(ours))
            {
                // A placeholder only where BC has nothing either: a real value already
                // in BC is better than a placeholder.
                if (f.Placeholder is not null && IsBcEmpty(theirs)) ours = f.Placeholder;
                else if (f.SkipWhenEmpty) continue;
            }

            // "" rather than null: BC clears a text field with an empty string.
            var a = (ours ?? string.Empty).Trim();
            var b = (theirs ?? string.Empty).Trim();

            if (!string.Equals(a, b, StringComparison.Ordinal))
                patch[f.BcField] = a;
        }

        return patch;
    }

    private string? Clip(string? value, int max, string number, string field)
    {
        if (value is null || value.Length <= max) return value;
        _logger.LogWarning(
            "Customer push: {Number} {Field} is {Length} chars, truncated to {Max}.",
            number, field, value.Length, max);
        return value[..max];
    }

    private static string Normalize(string? s) => (s ?? string.Empty).Trim();

    /// <summary>
    /// Pulls city, state and post code out of the last line of MailingAddress, which is
    /// free text in the shape "DALLAS, TX  75204".
    ///
    /// 341 of the 344 in-scope addresses match that shape, so parsing is worth doing --
    /// but it IS parsing, and the three that do not match simply come back empty rather
    /// than being guessed at. Leaving BC's fields untouched beats filling them wrongly.
    ///
    /// TaxingCity and TaxingState were considered as a cleaner source and rejected: they
    /// hold the JOB SITE's tax jurisdiction, not the customer's mailing address. Customer
    /// 80647 is Vancouver by post and Burnaby for tax.
    ///
    /// The post code pattern allows letters and an internal space so Canadian codes
    /// (V5K 1Z6) survive alongside US ZIP+4.
    /// </summary>
    private static (string? City, string? State, string? Zip) ParseCityStateZip(string? address)
    {
        if (string.IsNullOrWhiteSpace(address)) return (null, null, null);

        var parts = address.Replace("\r\n", "\n").Split('\n', StringSplitOptions.RemoveEmptyEntries)
                           .Select(p => p.Trim()).Where(p => p.Length > 0).ToList();
        if (parts.Count < 2) return (null, null, null);

        // The last line, not the second: a three-line address puts the suite on line 2.
        var m = CityStateZip.Match(parts[^1]);
        if (!m.Success) return (null, null, null);

        var zip = m.Groups[3].Value.Trim();
        return (m.Groups[1].Value.Trim(),
                m.Groups[2].Value.Trim(),
                zip.Length == 0 ? null : zip);
    }

    // CITY, ST [ZIP]. The zip is optional -- a few addresses stop at the state.
    private static readonly Regex CityStateZip = new(
        @"^(.+?),\s*([A-Za-z]{2})\s*([A-Za-z0-9][A-Za-z0-9\- ]*)?$",
        RegexOptions.Compiled);

    /// <summary>
    /// BC wants a two-letter country code. TaxingCountry is null on most rows and
    /// inconsistent where set ('USA' rather than 'US'), so it is normalised and anything
    /// unrecognised falls back to US -- which is right for all but a handful of Canadian
    /// customers, and those will be corrected by hand or by a later pull.
    /// </summary>
    private static string MapCountry(string? raw)
    {
        var s = raw?.Trim().ToUpperInvariant();
        return s switch
        {
            "CA" or "CAN" or "CANADA" => "CA",
            "MX" or "MEX" or "MEXICO" => "MX",
            _ => "US"
        };
    }

    /// <summary>
    /// EQData stores one address field with the lines separated by CRLF; BC wants them
    /// split. The baseline load built MailingAddress as addressLine1 + CRLF + addressLine2,
    /// so splitting on the first CRLF reverses it exactly for the common two-line case.
    ///
    /// A few source addresses run to three lines (street / suite / city-state-zip). BC has
    /// only two, so everything after the first line is joined into addressLine2. City,
    /// state and postalCode are deliberately NOT touched: EQData has no separate columns
    /// for them, and guessing them out of free text would be worse than leaving BC's.
    /// </summary>
    private static (string? Line1, string? Line2) SplitAddress(string? address)
    {
        if (string.IsNullOrWhiteSpace(address)) return (null, null);

        var parts = address.Replace("\r\n", "\n").Split('\n', StringSplitOptions.RemoveEmptyEntries)
                           .Select(p => p.Trim()).Where(p => p.Length > 0).ToList();

        if (parts.Count == 0) return (null, null);
        if (parts.Count == 1) return (parts[0], null);

        // BC has only two address lines; a three-line source address (street / suite /
        // city-state-zip) folds lines 2+ into addressLine2. City, state and postalCode
        // are left alone — EQData has no separate columns for them.
        return (parts[0], string.Join(", ", parts.Skip(1)));
    }

    private static string? Str(JsonElement el, string name) =>
        el.TryGetProperty(name, out var v) && v.ValueKind == JsonValueKind.String
            ? v.GetString()
            : null;

    /// <summary>Every mapped field, no comparison — there is nothing on the BC side to
    /// compare against yet.</summary>
    private Dictionary<string, object?> BuildCreate(PushField[] fields, Candidate row)
    {
        var body = new Dictionary<string, object?>();
        foreach (var f in fields)
        {
            var v = Clip(f.Read(row), f.MaxLength, row.CustomerID, f.BcField);
            if (string.IsNullOrWhiteSpace(v)) v = f.Placeholder;
            if (!string.IsNullOrWhiteSpace(v)) body[f.BcField] = v.Trim();
        }
        return body;
    }

    /// <summary>
    /// Creates a record and returns BC's assigned id, or null if nothing was written.
    ///
    /// The id is stored for convenience only. New-vs-existing is decided by whether BC
    /// returned a record with this number, never by whether BcId is populated — so a
    /// lost id costs a wasted lookup next run, not a duplicate.
    /// </summary>
    private async Task<Guid?> PostAsync(
        BcTarget target, string entity, Dictionary<string, object?> body,
        bool dryRun, string number, CancellationToken ct)
    {
        var json = JsonSerializer.Serialize(body);

        if (dryRun)
        {
            _logger.LogInformation("DRY RUN would POST {Entity} {Number}: {Payload}", entity, number, json);
            return null;
        }

        var http = _httpFactory.CreateClient();
        var token = await _tokens.GetTokenAsync(ct);

        using var req = new HttpRequestMessage(HttpMethod.Post, $"{target.ApiBase}/{entity}");
        req.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
        req.Content = new StringContent(json, Encoding.UTF8, "application/json");

        using var res = await http.SendAsync(req, ct);
        var responseBody = await res.Content.ReadAsStringAsync(ct);

        if (!res.IsSuccessStatusCode)
        {
            // Not fatal to the run: one record failing to create should not stop the rest.
            // Expect 400s here for required fields BC wants that we are not sending yet.
            _logger.LogError(
                "POST {Entity} {Number} failed ({Status}): {Body}",
                entity, number, (int)res.StatusCode, responseBody);
            return null;
        }

        using var doc = JsonDocument.Parse(responseBody);
        var id = Str(doc.RootElement, "id");
        if (id is null || !Guid.TryParse(id, out var guid))
        {
            _logger.LogWarning(
                "Created {Entity} {Number} but BC returned no usable id; it will be matched by number next run.",
                entity, number);
            return null;
        }

        _logger.LogInformation("Created {Entity} {Number} ({Id}): {Payload}", entity, number, guid, json);
        return guid;
    }

    private async Task StoreBcIdAsync(string customerId, Guid bcId, CancellationToken ct)
    {
        var sql = $@"
UPDATE {SourceTable}
   SET [BcId] = @BcId, [BcNumber] = @CustomerID, [LastSyncedAt] = sysutcdatetime()
 WHERE [CustomerID] = @CustomerID;";

        try
        {
            using var conn = new SqlConnection(_connectionString);
            await conn.ExecuteAsync(new CommandDefinition(sql,
                new { CustomerID = customerId, BcId = bcId }, cancellationToken: ct));
        }
        catch (Exception ex)
        {
            // Harmless: the next run matches on number and fills this in.
            _logger.LogWarning(ex, "Could not store BcId for {Number}.", customerId);
        }
    }
}
